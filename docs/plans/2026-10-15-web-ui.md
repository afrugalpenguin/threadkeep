# Web UI Implementation Plan

> **For the implementer:** REQUIRED SUB-SKILL: use superpowers:executing-plans (or subagent-driven-development) to implement this plan task by task.

**Goal:** Ship `packages/server` (Fastify JSON API plus a Preact single page) and a `hansard ui` command, per `docs/plans/2026-10-15-web-ui-design.md`.

**Architecture:** Core gains decision, facet and paging queries. The server is a thin Fastify app over those queries, guarded by a Host check and a per-run write token, serving an esbuild-bundled Preact client. The CLI imports the server lazily.

**Tech stack:** TypeScript (strict, NodeNext, `exactOptionalPropertyTypes`), Fastify 5, Preact 10, esbuild, Vitest, better-sqlite3.

**Rules for every task**

- TDD: write the failing test, run it and watch it fail, implement, run it and watch it pass, commit.
- Tests use synthetic fixtures in temp dirs only. Never touch `~/.claude`, `~/.hansard` or the real home dir.
- Fake secrets are assembled at runtime (see `packages/core/test/redaction.test.ts`) so scanners never see a literal token.
- SQL keywords are lowercase. No em or en dashes anywhere. Errors and logs carry ids, display paths and counts, never content or absolute home paths.
- Commit with a conventional subject (<= 72 chars) and the repo's `Co-Authored-By` trailer. The lefthook pre-commit and commit-msg hooks must pass; never bypass them.
- Run one test file with `pnpm vitest run <path>`. Run everything with `pnpm test`.

```ts
import type { Db } from './store/db.js';
import { NotFoundError, UsageError } from './query.js';

export const DECISION_STATUSES = ['proposed', 'confirmed', 'dismissed'] as const;
export type DecisionStatus = (typeof DECISION_STATUSES)[number];

export interface DecisionRow {
  id: string;
  session_id: string;
  event_id: string | null;
  statement: string;
  rationale: string | null;
  alternatives: string | null;
  status: DecisionStatus;
  machine_id: string | null;
  created_at: string | null;
  session_title: string | null;
}

export interface ListDecisionsOptions {
  status?: string;
  session?: string;
  limit?: number;
}

const columns = `d.id, d.session_id, d.event_id, d.statement, d.rationale, d.alternatives, d.status, d.machine_id, d.created_at,
  s.title as session_title`;

// Decisions are only visible through a non-excluded session.
export function listDecisions(db: Db, options: ListDecisionsOptions = {}): DecisionRow[] {
  const where = ['s.excluded = 0'];
  const params: (string | number)[] = [];
  if (options.status !== undefined) {
    if (!(DECISION_STATUSES as readonly string[]).includes(options.status)) {
      throw new UsageError(`invalid status: use ${DECISION_STATUSES.join(', ')}`);
    }
    where.push('d.status = ?');
    params.push(options.status);
  }
  if (options.session !== undefined) (where.push('d.session_id = ?'), params.push(options.session));
  params.push(options.limit ?? 200);
  return db
    .prepare(
      `select ${columns} from decisions d join sessions s on s.id = d.session_id
       where ${where.join(' and ')} order by d.created_at desc, d.id limit ?`,
    )
    .all(...params) as DecisionRow[];
}

export function setDecisionStatus(db: Db, id: string, status: 'confirmed' | 'dismissed'): DecisionRow {
  if (status !== 'confirmed' && status !== 'dismissed') throw new UsageError('status must be confirmed or dismissed');
  return db.transaction(() => {
    const found = db
      .prepare('select d.status from decisions d join sessions s on s.id = d.session_id where d.id = ? and s.excluded = 0')
      .get(id) as { status: DecisionStatus } | undefined;
    if (!found) throw new NotFoundError('decision not found');
    if (found.status !== 'proposed') throw new UsageError(`decision is already ${found.status}`);
    db.prepare("update decisions set status = ? where id = ? and status = 'proposed'").run(status, id);
    return db
      .prepare(`select ${columns} from decisions d join sessions s on s.id = d.session_id where d.id = ?`)
      .get(id) as DecisionRow;
  })();
}
```

```ts
import fs from 'node:fs';
import path from 'node:path';
import { timingSafeEqual } from 'node:crypto';
import Fastify, { type FastifyInstance, type FastifyRequest } from 'fastify';
import type { AddressInfo } from 'node:net';
import { openDb, type HansardContext } from '@hansard/core';

export interface AppOptions {
  token: string;
  port: number;
  publicDir: string;
  log: (line: string) => void;
}

const CSP = [
  "default-src 'none'",
  "script-src 'self'",
  "style-src 'self'",
  "connect-src 'self'",
  "img-src 'self' data:",
  "base-uri 'none'",
  "form-action 'none'",
  "frame-ancestors 'none'",
].join('; ');

const STATIC: Record<string, string> = {
  '/app.js': 'text/javascript; charset=utf-8',
  '/app.css': 'text/css; charset=utf-8',
};

export const indexHtml = (token: string): string =>
  `<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<meta name="hansard-token" content="${token}">
<title>Hansard</title>
<link rel="stylesheet" href="/app.css">
</head>
<body>
<div id="app"></div>
<script type="module" src="/app.js"></script>
</body>
</html>
`;

export const routeOf = (request: FastifyRequest): string => request.routeOptions.url ?? 'unmatched';

export function tokenMatches(expected: string, given: unknown): boolean {
  if (typeof given !== 'string') return false;
  const a = Buffer.from(expected);
  const b = Buffer.from(given);
  return a.length === b.length && timingSafeEqual(a, b);
}

export function createApp(ctx: HansardContext, options: AppOptions): FastifyInstance {
  const app = Fastify({ logger: false });
  const db = openDb(ctx.dbPath);
  app.decorate('db', db);
  app.addHook('onClose', async () => db.close());

  const boundPort = () => (app.server.address() as AddressInfo | null)?.port ?? options.port;
  app.addHook('onRequest', async (request, reply) => {
    const port = boundPort();
    const host = request.headers.host ?? '';
    if (host !== `127.0.0.1:${port}` && host !== `localhost:${port}`) {
      return reply.code(403).send({ error: 'forbidden host' });
    }
    if (request.method !== 'GET' && request.method !== 'HEAD' && !tokenMatches(options.token, request.headers['x-hansard-token'])) {
      return reply.code(403).send({ error: 'missing or invalid token' });
    }
  });
  app.addHook('onSend', async (_request, reply, payload) => {
    reply.header('content-security-policy', CSP);
    reply.header('x-content-type-options', 'nosniff');
    reply.header('referrer-policy', 'no-referrer');
    reply.header('cache-control', 'no-store');
    return payload;
  });
  // Method, route template and status only: never the query string, which can hold search text.
  app.addHook('onResponse', async (request, reply) => {
    options.log(`${request.method} ${routeOf(request)} ${reply.statusCode}`);
  });

  app.get('/', async (_request, reply) => reply.type('text/html; charset=utf-8').send(indexHtml(options.token)));
  for (const [url, type] of Object.entries(STATIC)) {
    app.get(url, async (_request, reply) => {
      const file = path.join(options.publicDir, url.slice(1));
      if (!fs.existsSync(file)) return reply.code(404).send({ error: 'not found' });
      return reply.type(type).send(fs.readFileSync(file));
    });
```

```ts
export interface TurnEvent {
  id: string;
  role: string;
  kind: string;
  files_touched: string[];
}

export interface Turn<E extends TurnEvent> {
  index: number;
  events: E[];
}

// A turn starts at each user text message; anything before the first one is turn 0.
export function groupTurns<E extends TurnEvent>(events: E[]): Turn<E>[] {
  const turns: Turn<E>[] = [];
  for (const event of events) {
    const last = turns[turns.length - 1];
    if (!last || (event.role === 'user' && event.kind === 'text')) turns.push({ index: turns.length, events: [event] });
    else last.events.push(event);
  }
  return turns;
}

export function turnOfEvent(turns: Turn<TurnEvent>[], eventId: string): number {
  const found = turns.find((t) => t.events.some((e) => e.id === eventId));
  return found?.index ?? 0;
}

export interface FileTouch {
  file: string;
  count: number;
  firstTurn: number;
}

export function filesUpTo(turns: Turn<TurnEvent>[], upTo: number): FileTouch[] {
  const files = new Map<string, FileTouch>();
  for (const turn of turns.slice(0, upTo + 1)) {
    for (const event of turn.events) {
      for (const file of event.files_touched) {
        const seen = files.get(file);
        if (seen) seen.count += 1;
        else files.set(file, { file, count: 1, firstTurn: turn.index });
      }
    }
  }
  return [...files.values()].sort((a, b) => b.count - a.count || a.file.localeCompare(b.file));
}
```

```ts
export type Params = Record<string, string>;
export type Route =
  | { view: 'sessions'; params: Params }
  | { view: 'search'; params: Params }
  | { view: 'session'; id: string; params: Params }
  | { view: 'decisions'; params: Params };

const fallback: Route = { view: 'sessions', params: {} };

export function parseHash(hash: string): Route {
  const [pathPart = '', query = ''] = hash.replace(/^#/, '').split('?', 2);
  const params: Params = Object.fromEntries(new URLSearchParams(query));
  const parts = pathPart.split('/').filter(Boolean).map(decodeURIComponent);
  const [view, id] = parts;
  if (view === 'sessions' || view === 'search' || view === 'decisions') return { view, params };
  if (view === 'session' && id) return { view, id, params };
  return { ...fallback, params: {} };
}

export function buildHash(route: Route): string {
  const query = new URLSearchParams(Object.entries(route.params).filter(([, v]) => v !== '')).toString();
  const base = route.view === 'session' ? `#/session/${encodeURIComponent(route.id)}` : `#/${route.view}`;
  return query ? `${base}?${query}` : base;
}
```

```ts
export class ApiError extends Error {
  constructor(readonly status: number, message: string) {
    super(message);
  }
}

async function call<T>(path: string, params: Record<string, string> = {}, init?: RequestInit): Promise<T> {
  const query = new URLSearchParams(Object.entries(params).filter(([, v]) => v !== '')).toString();
  const res = await fetch(query ? `${path}?${query}` : path, init);
  const body = (await res.json().catch(() => ({}))) as { error?: string };
  if (!res.ok) throw new ApiError(res.status, body.error ?? `request failed (${res.status})`);
  return body as T;
}

export const api = {
  sessions: (p: Record<string, string>) => call<SessionRow[]>('/api/sessions', p),
  search: (p: Record<string, string>) => call<SearchHit[]>('/api/search', p),
  session: (id: string, p: Record<string, string>) => call<ShownSession>(`/api/sessions/${encodeURIComponent(id)}`, p),
  facets: () => call<Facets>('/api/facets'),
  decisions: (p: Record<string, string>) => call<DecisionRow[]>('/api/decisions', p),
  setDecision: (id: string, status: 'confirmed' | 'dismissed') =>
    call<DecisionRow>(`/api/decisions/${encodeURIComponent(id)}/status`, {}, {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'x-hansard-token': token },
      body: JSON.stringify({ status }),
    }),
};
```

**Step 3: implement `packages/cli/src/commands/ui.ts`.**

```ts
import { spawn } from 'node:child_process';
import { displayPath } from '@hansard/core';
import { intFlag, parse, str } from '../args.js';
import { ExitError, usage, type Command } from '../types.js';

function openBrowser(url: string, onError: () => void): void {
  const [cmd, args] =
    process.platform === 'win32' ? ['cmd', ['/c', 'start', '""', url]] : process.platform === 'darwin' ? ['open', [url]] : ['xdg-open', [url]];
  try {
    const child = spawn(cmd, args, { detached: true, stdio: 'ignore', windowsHide: true });
    child.on('error', onError);
    child.unref();
  } catch {
    onError();
  }
}

export const uiCommand: Command = async (args, env) => {
  const { values, positionals } = parse(args, { port: { type: 'string' }, open: { type: 'boolean' } });
  if (positionals.length > 0) throw usage('ui takes no positional arguments');
  const port = intFlag(values['port'], 'port', 4477);
  if (port > 65535) throw usage('--port must be at most 65535');
  const ctx = env.context(str(values['home']));
  const { startServer } = await import('@hansard/server');
  let server;
  try {
    server = await startServer(ctx, { port, log: (line) => env.io.stderr.write(`${line}\n`) });
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === 'EADDRINUSE') throw new ExitError(`port ${port} is in use, pick another with --port`, 1);
    throw error;
  }
  env.io.stdout.write(`listening on ${server.url}\ndatabase: ${displayPath(ctx, ctx.dbPath)}\npress ctrl+c to stop\n`);
  if (values['open']) openBrowser(server.url, () => env.io.stderr.write('could not open a browser, visit the url above\n'));
  await new Promise<void>((resolve) => {
    if (env.signal?.aborted) return resolve();
    env.signal?.addEventListener('abort', () => resolve(), { once: true });
  });
  await server.close();
  return 0;
};
```
